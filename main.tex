% 无编号章：格式同 \BookChapter，但不出现"第 X 章"
\newcommand{\UnnumberedChapter}[2]{%
    \ifbilingual
        \chapter*{{\itshape #2}\\[4pt]#1}%
        \addcontentsline{toc}{chapter}{#2 / #1}%
    \else
        \chapter*{#1}%
        \addcontentsline{toc}{chapter}{#1}%
    \fi
}

% 无编号 section：格式同 \BookSection，但不出现编号
\newcommand{\UnnumberedSection}[2]{%
    \ifbilingual
        \section*{{\itshape #2}\\[2pt]#1}%
        \addcontentsline{toc}{section}{#2 / #1}%
    \else
        \section*{#1}%
        \addcontentsline{toc}{section}{#1}%
    \fi
}

% 10. 中英上下对照正文：\bi{中文}{English} 输出英文在上、中文在下
\newenvironment{bilingualtext}{}{}
\newcommand{\bi}[2]{%
    \ifbilingual
        #2\par
        \vspace{0.15em}%
        #1\par
        \vspace{0.9em}%
    \else
        #1\par
        \vspace{0.4em}%
    \fi
}

% 11. 中英双语 chapter 标题：英文斜体在上、中文加粗在下；目录显示 "English / 中文"
\newcommand{\BookChapter}[2]{%
    \ifbilingual
        \chapter[#2 / #1]{{\itshape #2}\\[4pt]#1}%
    \else
        \chapter[#1]{#1}%
    \fi
}

% 12. 中英双语 section 标题：英文斜体在上、中文加粗在下；目录显示 "English / 中文"
\renewcommand{\thesection}{\arabic{section}}

\newcommand{\BookSection}[2]{%
    \ifbilingual
        \section[#2 / #1]{{\itshape #2}\\[2pt]#1}%
    \else
        \section[#1]{#1}%
    \fi
}

\newcommand{\Bignote}[2]{%
    \ifbilingual
        \noindent\textbf{#2 \\ #1}%
    \else
        \noindent\textbf{#1}%
    \fi
}
